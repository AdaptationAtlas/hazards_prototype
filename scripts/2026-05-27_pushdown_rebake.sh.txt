Rscript - <<'EOF'
source("R/0_server_setup.R")
suppressPackageStartupMessages({pacman::p_load(DBI, duckdb)})

# Adjust path glob to the actual output dir of R/1.2 on this host
files <- Sys.glob(file.path(
  working_dir, "Data", "*", "*ensemble_season_timeseries*", "*", "*", "*ensemble_season_timeseries.parquet"
))
if (length(files) == 0L) {
  files <- Sys.glob(file.path(
    working_dir, "**", "*ensemble_season_timeseries.parquet"
  ))
}
cat("Found", length(files), "ensemble_season_timeseries.parquet files\n")
for (f in files) cat("  ", f, "\n")

con <- DBI::dbConnect(duckdb::duckdb())
for (f in files) {
  md <- DBI::dbGetQuery(con, sprintf(
    "SELECT COUNT(DISTINCT row_group_id) AS n_groups,
            COUNT(*) AS n_chunks,
            SUM(total_compressed_size)/1024 AS bytes_kb
     FROM parquet_metadata('%s')", f))
  iso3 <- DBI::dbGetQuery(con, sprintf(
    "SELECT COUNT(*) AS null_count FROM parquet_metadata('%s')
     WHERE path_in_schema = 'iso3' AND (stats_min IS NULL OR stats_max IS NULL)", f))
  cat(sprintf("%-60s  rg=%-4d  chunks=%-5d  avg=%.1f KB  iso3_null_stats=%d\n",
              basename(dirname(dirname(f))) , md$n_groups, md$n_chunks,
              md$bytes_kb / md$n_chunks, iso3$null_count))
}
DBI::dbDisconnect(con, shutdown = TRUE)
EOF

# -----------------------------------------------------------------------
# STAGE 6  promote sandbox → canonical  (gated on STAGE 5 pass)
# -----------------------------------------------------------------------
# WARNING: This rewrites canonical files. Backup pattern: download
# canonical, re-upload with `.preFix.bak` suffix, then move sandbox
# to canonical. Use download+upload (NOT s3_file_copy — strips ACL,
# see feedback_s3fs_copy_does_not_preserve_acl).

Rscript - <<'EOF'
source("R/0_server_setup.R")
suppressPackageStartupMessages({pacman::p_load(s3fs)})
s3fs::s3_credentials_set(
  access_key_id     = Sys.getenv("AWS_ACCESS_KEY_ID"),
  secret_access_key = Sys.getenv("AWS_SECRET_ACCESS_KEY")
)
BUCKET <- "digital-atlas"
CANON_PREFIX <- "domain=climate/type=hazard-indices/source=nex-gddp-cmip6/region=africa/processing=timeseries_mean_month/timeframe=3months"
periods <- c("1995-2014", "2021-2040", "2041-2060", "2061-2080", "2081-2100")
baseline <- "1995-2014"
tmp <- tempfile(fileext = ".parquet")
for (p in periods) {
  key <- sprintf("%s/period=%s/baseline=%s/variable=ensemble_season_timeseries.parquet",
                 CANON_PREFIX, p, baseline)
  canon  <- sprintf("s3://%s/%s", BUCKET, key)
  bak    <- sprintf("s3://%s/%s.preFix.bak", BUCKET, key)
  sandbx <- sprintf("s3://%s/sandbox/parquet-pushdown/%s", BUCKET, key)
  cat("BACKUP", canon, "->", bak, "\n")
  s3fs::s3_file_download(canon, tmp, overwrite = TRUE)
  s3fs::s3_file_upload(tmp, bak, ACL = "public-read")
  cat("PROMOTE", sandbx, "->", canon, "\n")
  s3fs::s3_file_download(sandbx, tmp, overwrite = TRUE)
  s3fs::s3_file_upload(tmp, canon, ACL = "public-read")
}
unlink(tmp)
EOF
