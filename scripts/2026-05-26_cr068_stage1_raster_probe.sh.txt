# ======================================================================
# 1B) Saturation probe — historic vs future, ONE GCM, per variant
# ======================================================================
# Probes one representative GCM (PROBE_GCM) for the four hazard families
# the categorisation step actually uses (NDWS, NTx35, NDWL0, THI-max —
# per the validator's `hazard_vars IN (...)` filter). For each variant
# (G-threshold), compares historic mean vs future mean. A historic-only
# saturation (high mean historic, normal future) is CR-068(b)'s smoking
# gun.
echo "" | tee -a "$LOG_FILE"
echo "===== 1B) Per-variant saturation probe (PROBE_GCM = $PROBE_GCM) =====" | tee -a "$LOG_FILE"
Rscript - <<'EOF' 2>&1 | tee -a "$LOG_FILE"
suppressPackageStartupMessages({library(terra); library(data.table)})
suppressMessages(suppressWarnings(
  source(file.path(Sys.getenv("PROJECT_DIR"), "R/0_server_setup.R"))
))
class_dir <- file.path(atlas_dirs$data_dir$hazard_timeseries_class, Sys.getenv("TIMEFRAME"))
gcm  <- Sys.getenv("PROBE_GCM")
fs   <- Sys.getenv("FUTURE_SCENARIO")
fp   <- Sys.getenv("FUTURE_PERIOD")

# Hazard families that participate in CR-068's categorisation step
HAZARDS <- c("NDWS", "NTx35", "NDWL0", "THI-max")

# Fast stats via terra::global — operates on layers without pulling
# everything into R memory. Returns one row per layer.
probe <- function(path) {
  r <- suppressWarnings(try(terra::rast(path), silent = TRUE))
  if (inherits(r, "try-error")) return(NULL)
  g <- suppressWarnings(terra::global(r,
    fun = c("min","mean","max"), na.rm = TRUE))
  v <- suppressWarnings(terra::values(r, na.rm = TRUE))
  if (length(v) == 0L) return(NULL)
  v <- as.numeric(v)
  data.table(
    file        = basename(path),
    n_finite    = length(v),
    mean        = g$mean[1],
    min         = g$min[1],
    max         = g$max[1],
    pct_eq_max  = round(100 * mean(v == g$max[1]), 2),
    pct_eq_zero = round(100 * mean(v == 0), 2)
  )
}

flagged_all <- list()
for (haz in HAZARDS) {
  cat(sprintf("\n--- %s ---\n", haz))
  hist_re <- sprintf("^historic_%s_[^_]+_%s[^.]*\\.tif$",   gcm, haz)
  fut_re  <- sprintf("^%s_%s_%s_%s[^.]*\\.tif$", fs, gcm, fp, haz)
  hist_f <- list.files(class_dir, hist_re, full.names = TRUE)
  fut_f  <- list.files(class_dir, fut_re,  full.names = TRUE)
  cat(sprintf("  historic files: %d   future (%s %s) files: %d\n",
              length(hist_f), fs, fp, length(fut_f)))
  if (length(hist_f) == 0 || length(fut_f) == 0) {
    cat("  (skipping — one side empty)\n")
    next
  }

  # Variant = the <variable> portion (e.g. 'NDWS-mean-G15')
  to_variant <- function(f) {
    sub(sprintf("^[^_]+_%s_[^_]+_", gcm), "",
        sub("\\.tif$", "", basename(f)))
  }
  hist_dt <- rbindlist(lapply(hist_f, probe))
  fut_dt  <- rbindlist(lapply(fut_f,  probe))
  if (nrow(hist_dt) == 0 || nrow(fut_dt) == 0) {
    cat("  (skipping — no readable files)\n"); next
  }
  hist_dt[, variant := to_variant(file)]
  fut_dt[,  variant := to_variant(file)]

  merged <- merge(
    hist_dt[, .(variant, h_mean = mean, h_pct_max = pct_eq_max, h_max = max)],
    fut_dt[,  .(variant, f_mean = mean, f_pct_max = pct_eq_max, f_max = max)],
    by = "variant"
  )
  merged[, mean_gap := h_mean - f_mean]
  merged[, sat_gap  := h_pct_max - f_pct_max]
  setorder(merged, -mean_gap)
  cat(sprintf("  %d variants matched — top 10 by mean_gap (historic - future):\n",
              nrow(merged)))
  print(merged[1:min(10L, .N)])
  merged[, hazard := haz]
  flagged_all[[haz]] <- merged
}

# Verdict
cat("\n=== VERDICT (1B) ===\n")
all_dt <- rbindlist(flagged_all, fill = TRUE)
saveRDS(all_dt, file.path(Sys.getenv("LOG_DIR"),
        sprintf("cr068_stage1b_%s.rds", Sys.getenv("STAMP"))))
threshold_mean <- 0.30   # 30 pp gap in mean classification
threshold_sat  <- 50     # 50 pp gap in pct_eq_max
flagged <- all_dt[mean_gap > threshold_mean | sat_gap > threshold_sat]
if (nrow(flagged) == 0L) {
  cat("NO variant exceeds thresholds (mean_gap > 0.30 OR sat_gap > 50 pp).\n")
  cat("CR-068(b) is NOT explained by simple historic-layer saturation in the\n")
  cat("classified rasters. Stage 2 should look UPSTREAM of classification —\n")
  cat("either the threshold function in R/2_calculate_haz_freq.R or the raw\n")
  cat("indices at /home/jovyan/common_data/atlas_nex-gddp_hazards/cmip6/indices.\n")
} else {
  cat(nrow(flagged), " variant(s) flagged:\n", sep = "")
  print(flagged[order(-mean_gap), .(hazard, variant, h_mean, f_mean, mean_gap, h_pct_max, f_pct_max, sat_gap)])
  cat("\nStage 2 should walk upstream from these variants — inspect the raw\n")
  cat("indices feeding them at indices_dir, plus the threshold step in\n")
  cat("R/2_calculate_haz_freq.R that emits the classified values.\n")
}
EOF


# ======================================================================
# 1C) SSP370 raw-input coverage — CR-068(c) bundled probe
# ======================================================================
# Raw NEX-GDDP indices live at indices_dir (NOT under Data/). Walk the
# tree for each (scenario, period) combo we'd expect and count files,
# disambiguating whether SSP370 future is missing because the fetch
# pipeline didn't pull it (raw inputs absent) or because Step 2 dropped
# it (raw inputs present but classification emits nothing).
echo "" | tee -a "$LOG_FILE"
echo "===== 1C) SSP370 raw-input coverage at indices_dir =====" | tee -a "$LOG_FILE"
Rscript - <<'EOF' 2>&1 | tee -a "$LOG_FILE"
suppressPackageStartupMessages({library(data.table)})
suppressMessages(suppressWarnings(
  source(file.path(Sys.getenv("PROJECT_DIR"), "R/0_server_setup.R"))
))
if (!exists("indices_dir") || !dir.exists(indices_dir)) {
  stop("indices_dir not set or missing — check R/0_server_setup.R lines 150-158")
}
cat("indices_dir = ", indices_dir, "\n\n", sep = "")

# Walk depth=3 so we see scenario / period / gcm or scenario / gcm /
# period or similar — print the tree skeleton first.
top <- list.dirs(indices_dir, recursive = FALSE, full.names = FALSE)
cat("Top-level subdirs:\n  ", paste(top, collapse = "\n  "), "\n\n", sep = "")

# Search for TIFs anywhere and parse path to (scenario, period)
all_tifs <- list.files(indices_dir, "\\.tif$",
                       recursive = TRUE, full.names = FALSE)
cat("Total indices TIFs found: ", length(all_tifs), "\n", sep = "")
if (length(all_tifs) == 0L) {
  cat("(no TIFs at all under indices_dir — bigger problem upstream)\n")
} else {
  # Extract scenario tokens (ssp*, historic) and period tokens (YYYY-YYYY)
  paths <- data.table(path = all_tifs)
  paths[, scenario := regmatches(path,
    regexpr("ssp[0-9]+|historic", path, ignore.case = TRUE))]
  paths[, period := regmatches(path,
    regexpr("[0-9]{4}-[0-9]{4}", path))]
  paths[is.na(scenario) | scenario == "", scenario := "unknown"]
  paths[is.na(period)   | period == "",   period := "unknown"]

  cat("\nFile counts per (scenario, period):\n")
  print(paths[, .N, by = .(scenario, period)][order(scenario, period)])

  cat("\nFile counts per scenario only:\n")
  print(paths[, .N, by = scenario][order(scenario)])
}

cat("\n=== VERDICT (1C) ===\n")
periods_expected <- c("2021-2040","2041-2060","2061-2080","2081-2100")
if (length(all_tifs) == 0L) {
  cat("indices_dir empty → upstream fetch may have been deleted or moved.\n")
} else {
  # Compare ssp370 future coverage vs ssp245
  ssp370_periods <- paths[scenario == "ssp370", unique(period)]
  ssp245_periods <- paths[scenario == "ssp245", unique(period)]
  missing_370 <- setdiff(periods_expected, ssp370_periods)
  present_245 <- intersect(periods_expected, ssp245_periods)
  cat("SSP245 future periods present at indices_dir: ",
      paste(present_245, collapse = ", "), "\n", sep = "")
  cat("SSP370 future periods present at indices_dir: ",
      paste(intersect(periods_expected, ssp370_periods), collapse = ", "), "\n", sep = "")
  cat("SSP370 future periods MISSING at indices_dir: ",
      paste(missing_370, collapse = ", "), "\n", sep = "")
  if (length(missing_370) == 0L) {
    cat("\nSSP370 raw inputs PRESENT for all future periods.\n")
    cat("=> The 2041+ zero data we see in hazard_exposure is a Step 2\n")
    cat("   classification bug, NOT a fetch bug. Inspect R/2_calculate_haz_freq.R\n")
    cat("   for scenario-dependent branches that drop SSP370.\n")
  } else {
    cat("\nSSP370 raw inputs MISSING for: ",
        paste(missing_370, collapse = ", "), "\n", sep = "")
    cat("=> The 2041+ zero data is an UPSTREAM FETCH bug. The NEX-GDDP fetch\n")
    cat("   pipeline did not pull SSP370 future from NASA. Need to re-pull\n")
    cat("   then re-run R/1_make_timeseries.R for those periods.\n")
  }
}
EOF
