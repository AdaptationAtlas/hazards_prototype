# ======================================================================
# 1B) Per-hazard saturation probe — HISTORIC
# ======================================================================
# For each classified hazard raster under hazard_timeseries_class/historic/,
# compute summary stats. The "saturation" signal is mean ~ 1 AND
# fraction_of_pixels_eq_max ~ 1 (every cell exceeds the threshold
# every year, so the layer carries no information). If multiple
# layers (bands), aggregate per-band first.
#
# Stats reported per raster:
#   n_layers, n_pixels, n_finite, min, p05, p50, mean, p95, max,
#   pct_eq_max (saturation), pct_eq_zero (dead pixels)
echo "" | tee -a "$LOG_FILE"
echo "===== 1B) Historic saturation probe =====" | tee -a "$LOG_FILE"
Rscript - <<'EOF' 2>&1 | tee -a "$LOG_FILE"
suppressPackageStartupMessages({library(terra); library(data.table)})
suppressMessages(suppressWarnings(
  source(file.path(Sys.getenv("PROJECT_DIR"), "R/0_server_setup.R"))
))
class_root <- atlas_dirs$data_dir$hazard_timeseries_class
hist_dir <- file.path(class_root, "historic")
if (!dir.exists(hist_dir)) {
  stop("historic dir missing at ", hist_dir,
       " — check Stage 1A inventory for the actual name (could be 'hist', '1995-2014', etc.)")
}

probe_raster <- function(f) {
  r <- tryCatch(terra::rast(f), error = function(e) NULL)
  if (is.null(r)) return(data.table(file = basename(f), error = "rast() failed"))
  v <- terra::values(r, dataframe = FALSE, na.rm = TRUE)
  if (length(v) == 0L) return(data.table(file = basename(f), error = "empty"))
  v <- as.numeric(v)
  qs <- quantile(v, c(0.05, 0.50, 0.95), na.rm = TRUE, names = FALSE)
  data.table(
    file       = basename(f),
    n_layers   = terra::nlyr(r),
    n_finite   = length(v),
    min        = min(v, na.rm = TRUE),
    p05        = qs[1],
    p50        = qs[2],
    mean       = mean(v, na.rm = TRUE),
    p95        = qs[3],
    max        = max(v, na.rm = TRUE),
    pct_eq_max = round(100 * mean(v == max(v, na.rm = TRUE), na.rm = TRUE), 2),
    pct_eq_zero = round(100 * mean(v == 0, na.rm = TRUE), 2)
  )
}

files <- list.files(hist_dir, pattern = "\\.tif$", full.names = TRUE)
cat("Found ", length(files), " classified rasters under ", hist_dir, "\n\n", sep = "")
if (length(files) == 0L) {
  cat("(no rasters — skipping)\n")
} else {
  # Probe all, but show a progress dot per 5 files (some folders have 100s)
  out <- vector("list", length(files))
  for (i in seq_along(files)) {
    out[[i]] <- probe_raster(files[[i]])
    if (i %% 5 == 0L) cat(".")
    if (i %% 200 == 0L) cat(" ", i, "\n", sep = "")
  }
  cat("\n")
  hist_stats <- rbindlist(out, fill = TRUE)
  hist_stats <- hist_stats[order(-pct_eq_max)]
  cat("\nTop 20 by pct_eq_max (saturation candidates first):\n")
  print(hist_stats[1:min(20, .N)])
  # Save full table for downstream comparison
  saveRDS(hist_stats, file.path(Sys.getenv("LOG_DIR"),
          sprintf("cr068_hist_stats_%s.rds", Sys.getenv("STAMP"))))
  cat("\nFull table saved to ", Sys.getenv("LOG_DIR"),
      "/cr068_hist_stats_", Sys.getenv("STAMP"), ".rds\n", sep = "")
}
EOF

# ======================================================================
# 1D) Diff verdict — which layers are saturated in historic but NOT future?
# ======================================================================
# Joins the two stats tables by file basename and ranks the largest
# (hist.mean - fut.mean) gaps. A layer with hist.mean ~ 1 (or
# hist.pct_eq_max ~ 100 %) AND fut.mean << 1 is the categorisation
# bug's smoking gun: the layer carries no information historically.
echo "" | tee -a "$LOG_FILE"
echo "===== 1D) Saturation diff: HISTORIC vs FUTURE =====" | tee -a "$LOG_FILE"
Rscript - <<'EOF' 2>&1 | tee -a "$LOG_FILE"
suppressPackageStartupMessages({library(data.table)})
log_dir <- Sys.getenv("LOG_DIR")
stamp   <- Sys.getenv("STAMP")
hist_rds <- file.path(log_dir, sprintf("cr068_hist_stats_%s.rds", stamp))
fut_rds  <- file.path(log_dir, sprintf("cr068_fut_stats_%s.rds",  stamp))
if (!file.exists(hist_rds) || !file.exists(fut_rds)) {
  stop("Stats RDS files missing — run 1B and 1C first.")
}
h <- readRDS(hist_rds); f <- readRDS(fut_rds)
setnames(h, c("mean", "pct_eq_max", "pct_eq_zero", "p50", "p95", "max"),
            c("h_mean","h_pct_max","h_pct_zero","h_p50","h_p95","h_max"))
setnames(f, c("mean", "pct_eq_max", "pct_eq_zero", "p50", "p95", "max"),
            c("f_mean","f_pct_max","f_pct_zero","f_p50","f_p95","f_max"))
keep <- c("file","h_mean","h_pct_max","h_pct_zero","f_mean","f_pct_max","f_pct_zero")
merged <- merge(h[, ..keep[c(1:4)]], f[, ..keep[c(1, 5:7)]], by = "file", all = FALSE)
merged[, mean_gap := h_mean - f_mean]
merged[, sat_gap  := h_pct_max - f_pct_max]
setorder(merged, -mean_gap)
cat("Top 20 layers by hist.mean - fut.mean (positive => historic saturated more):\n")
print(merged[1:min(20, .N)])

cat("\n--- Verdict ---\n")
threshold_mean_gap <- 0.30   # 30+ pp difference in mean classification
threshold_sat_gap  <- 50     # 50+ pp difference in pct_eq_max
flagged <- merged[mean_gap > threshold_mean_gap | sat_gap > threshold_sat_gap]
if (nrow(flagged) == 0L) {
  cat("NO clear saturation gap. CR-068(b) categorisation asymmetry may NOT be ",
      "explained by simple historic-layer saturation — the bug is elsewhere ",
      "(e.g. threshold function, layer rename collision, scenario-dependent ",
      "categorisation). Stage 2 should walk upstream (raw timeseries + ",
      "thresholds) rather than re-bake the classified rasters.\n", sep = "")
} else {
  cat(nrow(flagged), " layer(s) flagged as potentially saturated historic:\n\n", sep = "")
  print(flagged[order(-mean_gap)])
  cat("\nLikely root-cause direction: the raw timeseries that feed these layers ",
      "produce saturated counts under the historic baseline. Stage 2 should ",
      "inspect Data/hazard_timeseries/historic/ for the parent raster of each ",
      "flagged layer + the threshold function that emits the classified value.\n", sep = "")
}
EOF

# Walk the directory two levels deep so we see scenario × period combos
list_tree <- function(root, depth = 3) {
  d <- list.dirs(root, recursive = TRUE, full.names = TRUE)
  rel <- sub(paste0("^", root, "/?"), "", d)
  parts <- strsplit(rel, "/")
  keep <- vapply(parts, function(p) length(p) > 0L && length(p) <= depth, logical(1))
  d[keep]
}
all_dirs <- list_tree(ts_root, depth = 3)
cat("Subdirs (depth <=3):\n")
for (d in all_dirs) cat("  ", sub(paste0("^", ts_root, "/?"), "", d), "\n", sep = "")

# Per (scenario, period), count TIFs — empty cells = no data
cat("\nFile counts per (scenario, period):\n")
periods   <- c("2021-2040","2041-2060","2061-2080","2081-2100")
scenarios <- c("ssp126","ssp245","ssp370","ssp585")
results <- data.table(
  scenario = rep(scenarios, each = length(periods)),
  period   = rep(periods,    times = length(scenarios)),
  n_tifs   = NA_integer_,
  exists   = NA
)
for (i in seq_len(nrow(results))) {
  # Probe a couple of plausible layouts: <scenario>/<period> AND <period>/<scenario>
  cand1 <- file.path(ts_root, results$scenario[i], results$period[i])
  cand2 <- file.path(ts_root, results$period[i],   results$scenario[i])
  cand3 <- file.path(ts_root, paste0(results$scenario[i], "_", results$period[i]))
  found <- c(cand1, cand2, cand3)[c(dir.exists(cand1), dir.exists(cand2), dir.exists(cand3))]
  results$exists[i] <- length(found) > 0L
  results$n_tifs[i] <- if (length(found) == 0L) 0L else
    length(list.files(found[1], pattern = "\\.tif$", recursive = TRUE))
}
print(results)

cat("\n--- Verdict ---\n")
ssp370_present <- results[scenario == "ssp370" & n_tifs > 0L, .N]
ssp245_present <- results[scenario == "ssp245" & n_tifs > 0L, .N]
