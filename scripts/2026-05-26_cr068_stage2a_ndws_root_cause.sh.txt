# Helper: read first band only, compute fast global stats. (Some
# indices files are multi-band yearly stacks; we still want a
# single-band summary so a saturated layer is visible.)
probe <- function(path) {
  r <- suppressWarnings(try(terra::rast(path), silent = TRUE))
  if (inherits(r, "try-error")) return(NULL)
  nl <- terra::nlyr(r)
  # First-band stats
  r1 <- if (nl > 1L) r[[1]] else r
  v <- suppressWarnings(as.numeric(terra::values(r1, na.rm = TRUE)))
  if (length(v) == 0L) return(NULL)
  qs <- quantile(v, c(0.05, 0.50, 0.95), na.rm = TRUE, names = FALSE)
  # Also probe per-band MEAN if multi-band (so we see annual trend)
  per_band_means <- NA_real_
  if (nl > 1L) {
    g <- suppressWarnings(terra::global(r, fun = "mean", na.rm = TRUE))
    per_band_means <- paste(round(g$mean, 1), collapse = ",")
  }
  data.table(
    file       = basename(path),
    n_layers   = nl,
    band1_n    = length(v),
    band1_min  = min(v, na.rm = TRUE),
    band1_p05  = qs[1],
    band1_p50  = qs[2],
    band1_mean = mean(v, na.rm = TRUE),
    band1_p95  = qs[3],
    band1_max  = max(v, na.rm = TRUE),
    per_band_means = per_band_means
  )
}

run_dir <- function(label, dir) {
  cat(sprintf("\n--- %s — %s ---\n", label, dir))
  if (!dir.exists(dir)) { cat("(dir missing)\n"); return(NULL) }
  files <- list.files(dir, pattern = "NDWS", recursive = TRUE,
                      full.names = TRUE)
  if (length(files) == 0L) { cat("(no NDWS files)\n"); return(NULL) }
  cat("Found ", length(files), " NDWS file(s). Probing first 10:\n",
      sep = "")
  files <- head(files, 10)
  out <- rbindlist(lapply(files, probe), fill = TRUE)
  print(out)
  invisible(out)
}

hist_stats <- run_dir("HISTORIC raw NDWS", hist_subdir)
fut_stats  <- run_dir("FUTURE raw NDWS",   fut_subdir)

cat("\n=== VERDICT (2A.2) ===\n")
if (is.null(hist_stats) || is.null(fut_stats)) {
  cat("Cannot diff — one or both NDWS sets are absent.\n")
  cat("Re-check 2A.1's inventory: NDWS may live under a different ",
      "filename convention (e.g. NDWS-G15.tif as the index-name not ",
      "just NDWS).\n", sep = "")
} else {
  hist_mean <- mean(hist_stats$band1_mean, na.rm = TRUE)
  fut_mean  <- mean(fut_stats$band1_mean,  na.rm = TRUE)
  hist_max  <- max(hist_stats$band1_max,  na.rm = TRUE)
  fut_max   <- max(fut_stats$band1_max,   na.rm = TRUE)
  # Saturation ratio = mean / max — scale-free, works for daily/monthly/annual
  hist_sat <- hist_mean / hist_max
  fut_sat  <- fut_mean  / fut_max
  cat(sprintf("Aggregate raw-NDWS band-1 mean:  historic=%.2f  future=%.2f  gap=%.2f\n",
              hist_mean, fut_mean, hist_mean - fut_mean))
  cat(sprintf("Aggregate raw-NDWS band-1 max:   historic=%.2f  future=%.2f\n",
              hist_max, fut_max))
  cat(sprintf("Saturation ratio (mean/max):     historic=%.3f  future=%.3f  gap=%.3f\n",
              hist_sat, fut_sat, hist_sat - fut_sat))
  if (hist_sat > 0.85 && fut_sat < 0.85 && hist_mean > fut_mean) {
    cat("\nHistoric raw NDWS is SATURATED (mean/max > 0.85) while future\n")
    cat("retains normal variance. The bug is UPSTREAM of classification —\n")
    cat("investigate the upstream `hazards` repo at\n")
    cat("https://github.com/AdaptationAtlas/hazards (daily PET/AWC + NDWS\n")
    cat("monthly aggregation). The classification logic in\n")
    cat("R/2_calculate_haz_freq.R is doing the right thing on broken inputs.\n")
    cat("Dispatch handed to that repo's maintainer:\n")
    cat("  atlas_notebooks/playbook/handovers/climateRationale/dispatches/\n")
    cat("    2026-05-26_hazards-repo-ndws-historic-saturation.md\n")
  } else if (abs(hist_sat - fut_sat) < 0.10) {
    cat("\nHistoric raw NDWS saturation ratio is SIMILAR to future\n")
    cat("(within 10 pp). The saturation arrives AT CLASSIFICATION, not\n")
    cat("upstream. Investigate the threshold logic in\n")
    cat("R/2_calculate_haz_freq.R for a scenario-dependent branch.\n")
  } else {
    cat("\nDirection intermediate — saturation gap exists but not as\n")
    cat("clear-cut as the canonical upstream-bug fingerprint. Inspect\n")
    cat("the per-band-means above for which years drive the gap, then\n")
    cat("escalate to Pete.\n")
  }
}
