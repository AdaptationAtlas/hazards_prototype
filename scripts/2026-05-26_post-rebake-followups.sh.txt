# ======================================================================
# A) NGA oilpalm = NaN — UNKNOWN, the only finding without a dispatch
# ======================================================================
# Validator [c] reported NaN for every (scenario, timeframe) at
# iso3='NGA', crop='oilpalm'. Three possibilities to discriminate:
#   A1. NGA has no oilpalm rows at all in the parquet
#       -> validator [c] spot-check is meaningless for this crop;
#          patch the validator to drop or swap it. NOT a publish blocker.
#   A2. NGA oilpalm rows exist but `value` is NaN end-to-end
#       -> NaN injected upstream; likely the MapSPAM oilpalm raster
#          has NaN over NGA (or zonal-sum saw zero non-NA cells).
#          Investigate the producer TIF before publish.
#   A3. Some NGA oilpalm rows have non-NA values, others are NaN
#       -> partial NaN; sum_value collapses to NaN. Treat as A2.
#
# Decision: A1 -> patch validator (no S3 blocker).
#           A2 / A3 -> open a follow-up dispatch + hold publish.
echo "" | tee -a "$LOG_FILE"
echo "===== A) NGA oilpalm row inventory =====" | tee -a "$LOG_FILE"
Rscript - <<'EOF' 2>&1 | tee -a "$LOG_FILE"
suppressPackageStartupMessages({library(duckdb); library(DBI)})
con <- dbConnect(duckdb())
parquet <- Sys.getenv("PARQUET")
sql <- sprintf("
  SELECT
    COUNT(*)                                                          AS n_rows,
    COUNT(*) FILTER (WHERE value IS NOT NULL AND NOT isnan(value))    AS n_finite,
    SUM(CASE WHEN value IS NULL THEN 1 ELSE 0 END)                    AS n_null,
    SUM(CASE WHEN value IS NOT NULL AND isnan(value) THEN 1 ELSE 0 END) AS n_nan,
    MIN(value) FILTER (WHERE value IS NOT NULL AND NOT isnan(value))  AS min_val,
    MAX(value) FILTER (WHERE value IS NOT NULL AND NOT isnan(value))  AS max_val
  FROM read_parquet('%s')
  WHERE iso3 = 'NGA' AND crop = 'oilpalm'
", parquet)
print(dbGetQuery(con, sql))
EOF

# If n_rows  == 0 -> outcome A1.  Drop / swap the spot-check.
# If n_nan   >  0 -> outcome A2 / A3.  Inspect the producer TIF next.

# Cross-check: is oilpalm produced anywhere else in the parquet?
echo "" | tee -a "$LOG_FILE"
echo "===== A.bis) global oilpalm coverage (any iso3) =====" | tee -a "$LOG_FILE"
Rscript - <<'EOF' 2>&1 | tee -a "$LOG_FILE"
suppressPackageStartupMessages({library(duckdb); library(DBI)})
con <- dbConnect(duckdb())
parquet <- Sys.getenv("PARQUET")
sql <- sprintf("
  SELECT iso3,
         COUNT(*)                                                          AS n_rows,
         SUM(CASE WHEN value IS NOT NULL AND isnan(value) THEN 1 ELSE 0 END) AS n_nan,
         COUNT(*) FILTER (WHERE value IS NOT NULL AND NOT isnan(value))    AS n_finite
  FROM read_parquet('%s')
  WHERE crop = 'oilpalm' AND admin1_name IS NULL AND admin2_name IS NULL
  GROUP BY iso3 ORDER BY iso3
", parquet)
print(dbGetQuery(con, sql))
EOF

# If oilpalm is widely populated EXCEPT for NGA -> the NGA-specific
# MapSPAM tif has a coverage hole (or the NGA admin0 mask misses it).
# If oilpalm is NaN across multiple iso3 -> rebake artefact, hold publish.


# ======================================================================
# B) SSP370 2041+ zeros — expected fingerprint of CR-068
# ======================================================================
# CR-068 SSP370 missing-periods signature (see
# R/checks/68_categorisation_stage1.R:289-315): SSP370 2021-2040 is
# populated; SSP370 2041-2060, 2061-2080, 2081-2100 are zero across
# every (iso3, crop). If our parquet matches that fingerprint we can
# confirm "known bug, no new regression". If 2021-2040 ALSO collapses
# to zero, or other scenarios drop out, that IS new and a blocker.
echo "" | tee -a "$LOG_FILE"
echo "===== B) SSP370 vs SSP245 per-period coverage =====" | tee -a "$LOG_FILE"
Rscript - <<'EOF' 2>&1 | tee -a "$LOG_FILE"
suppressPackageStartupMessages({library(duckdb); library(DBI)})
con <- dbConnect(duckdb())
parquet <- Sys.getenv("PARQUET")
sql <- sprintf("
  SELECT scenario, timeframe,
         COUNT(*)   AS n_rows,
         SUM(value) AS sum_value,
         SUM(CASE WHEN value = 0 THEN 1 ELSE 0 END) AS n_zero
  FROM read_parquet('%s')
  WHERE scenario IN ('ssp245','ssp370')
    AND hazard = 'any'
    AND admin1_name IS NULL AND admin2_name IS NULL
  GROUP BY scenario, timeframe
  ORDER BY scenario, timeframe
", parquet)
print(dbGetQuery(con, sql))
EOF

# Expected (CR-068 fingerprint):
#   ssp245 all 4 future periods: sum_value > 0
#   ssp370 2021-2040: sum_value > 0
#   ssp370 2041-2060 / 2061-2080 / 2081-2100: sum_value == 0
# If you see anything else, treat as a new regression and hold publish.


# ======================================================================
# C) AGO historic heat / wet / heat+wet = 0 — expected fingerprint

# ======================================================================

suppressPackageStartupMessages({library(duckdb); library(DBI)})
con <- dbConnect(duckdb())
parquet <- Sys.getenv("PARQUET")
sql <- sprintf("
  SELECT hazard, ROUND(SUM(value)::DOUBLE, 0) AS hist_sum
  FROM read_parquet('%s')
  WHERE iso3 = 'AGO' AND scenario = 'historic' AND timeframe = '1995-2014'
    AND hazard NOT IN ('none','any')
    AND admin1_name IS NULL AND admin2_name IS NULL
  GROUP BY hazard
  ORDER BY hist_sum DESC
", parquet)
print(dbGetQuery(con, sql))
EOF

# Expected (CR-068 fingerprint):
#   dry / dry+wet / dry+heat / dry+heat+wet: > 0
#   heat / wet / heat+wet: == 0
# If a "dry" combo drops to zero historically, that IS a #9 regression
# and a hard publish blocker.
